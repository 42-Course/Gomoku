1//! Heuristic evaluation from the side-to-move's perspective.
2//!
3//! Scores are positive when the position favors the player whose turn it is.
4//! The negamax search calls [`evaluate`] only at non-terminal nodes; the win
5//! shortcut here is just a safety net for callers outside the search.
6//!
7//! # Pipeline
8//!
9//! 1. [`Board::for_each_line`] yields every row, column, and diagonal as a
10//!    pair of `(me, opp)` bitmasks plus a length.
11//! 2. [`crate::patterns::count_patterns`] tallies the runs on each line by
12//!    length and openness.
13//! 3. [`score_from_counts`] converts the tallies to a single integer using
14//!    the per-pattern weights below.
15//!
16//! That's substantially cheaper than the per-cell walk this module used to
17//! do, especially under a deep search.
18//!
19//! # Sign convention
20//!
21//! [`evaluate`] returns the score from the perspective of
22//! [`Game::current_player`]; positive means the side to move is winning.
23//! This is the convention negamax expects — the caller negates between plies.
24
25use crate::board::Board;
26use crate::game::{Game, GameStatus, Player};
27use crate::patterns::{count_patterns, PatternCounts};
28
29/// Score returned for a winning terminal position. Mirrors
30/// [`crate::ai::search::WIN_SCORE`] so the two modules can be reasoned
31/// about independently.
32pub const WIN_SCORE: i32    = 1_000_000;
33const OPEN_FOUR: i32        = 100_000;
34const CLOSED_FOUR: i32      = 20_000;
35const OPEN_THREE: i32       = 5_000;
36const CLOSED_THREE: i32     = 1_000;
37const OPEN_TWO: i32         = 300;
38const CLOSED_TWO: i32       = 50;
39const CAPTURE_PAIR: i32     = 2_000;
40
41/// A single 5-in-a-row already wins, but we never reach this branch in
42/// search — `terminal_score` has the final word at the root. Five-counts
43/// here are a defensive fallback if eval is called from outside.
44const FIVE_FALLBACK: i32    = WIN_SCORE;
45
46/// Score the position from the side-to-move's perspective.
47///
48/// Positive means the player on move is favored, negative means the
49/// opponent is. Decisive positions return `±WIN_SCORE`; everything else
50/// is a weighted sum of pattern counts plus a capture-difference term.
51///
52/// # Examples
53///
54/// ```ignore
55/// let game = Game::new();
56/// assert_eq!(evaluate(&game), 0); // empty board is balanced
57/// ```
58pub fn evaluate(game: &Game) -> i32 {
59    if let GameStatus::Win(winner) = game.status {
60        return if winner == game.current_player() { WIN_SCORE } else { -WIN_SCORE };
61    }
62
63    let me = game.current_player();
64    let my_score = score_player(&game.board, me);
65    let opp_score = score_player(&game.board, me.opponent());
66    let capture_score = capture_diff(game, me) * CAPTURE_PAIR;
67
68    my_score - opp_score + capture_score
69}
70
71/// Capture pairs `me` is ahead by. Negative when `me` is behind.
72fn capture_diff(game: &Game, me: Player) -> i32 {
73    let (black, white) = (game.captures.0 as i32, game.captures.1 as i32);
74    match me {
75        Player::Black => black - white,
76        Player::White => white - black,
77    }
78}
79
80/// Total pattern score for one player across every line on the board.
81fn score_player(board: &Board, player: Player) -> i32 {
82    let mut totals = PatternCounts::default();
83    // No useful pattern fits in fewer than 5 cells — skip stub diagonals.
84    board.for_each_line(player, 5, |me, opp, len| {
85        let line = count_patterns(me, opp, len);
86        totals.add(&line);
87    });
88
89    score_from_counts(&totals)
90}
91
92/// Convert a [`PatternCounts`] tally to an integer score using the
93/// per-pattern weights at the top of this module.
94fn score_from_counts(c: &PatternCounts) -> i32 {
95    (c.fives as i32) * FIVE_FALLBACK
96        + (c.open_four as i32) * OPEN_FOUR
97        + (c.closed_four as i32) * CLOSED_FOUR
98        + (c.open_three as i32) * OPEN_THREE
99        + (c.closed_three as i32) * CLOSED_THREE
100        + (c.open_two as i32) * OPEN_TWO
101        + (c.closed_two as i32) * CLOSED_TWO
102}

104#[cfg(test)]
105mod tests {
106    use super::*;
107    use crate::game::Game;
108
109    #[test]
110    fn empty_board_is_balanced() {
111        let game = Game::new();
112        assert_eq!(evaluate(&game), 0);
113    }
114
115    #[test]
116    fn open_three_outscores_open_two() {
117        // Black builds an open three: . X X X .
118        let mut three = Game::new();
119        three.play_move(5, 9).unwrap();   // B
120        three.play_move(0, 0).unwrap();   // W (parking, far away)
121        three.play_move(6, 9).unwrap();   // B
122        three.play_move(0, 1).unwrap();   // W
123        three.play_move(7, 9).unwrap();   // B  → it's now White to move
124
125        // Same setup but only an open two: . X X .
126        let mut two = Game::new();
127        two.play_move(5, 9).unwrap();
128        two.play_move(0, 0).unwrap();
129        two.play_move(6, 9).unwrap();
130        two.play_move(0, 1).unwrap();
131        two.play_move(8, 0).unwrap();      // extra B so White is on move here too
132
133        assert!(
134            evaluate(&three) < evaluate(&two),
135            "open three (={}) should be more painful than open two (={})",
136            evaluate(&three),
137            evaluate(&two),
138        );
139    }
140
141    #[test]
142    fn capture_lead_helps_the_capturer() {
143        let mut game = Game::new();
144        game.play_move(0, 0).unwrap();   // B
145        game.play_move(1, 0).unwrap();   // W
146        game.play_move(4, 0).unwrap();   // B
147        game.play_move(2, 0).unwrap();   // W
148        game.play_move(3, 0).unwrap();   // B → captures (1,0) and (2,0)
149
150        assert_eq!(game.captures, (1, 0));
151        let score = evaluate(&game);
152        assert!(score <= -CAPTURE_PAIR, "expected capture penalty, got {score}");
153    }
154
155    #[test]
156    fn winning_status_dominates() {
157        let mut game = Game::new();
158        game.play_move(0, 0).unwrap();
159        game.play_move(0, 1).unwrap();
160        game.play_move(1, 0).unwrap();
161        game.play_move(1, 1).unwrap();
162        game.play_move(2, 0).unwrap();
163        game.play_move(2, 1).unwrap();
164        game.play_move(3, 0).unwrap();
165        game.play_move(3, 1).unwrap();
166        game.play_move(4, 0).unwrap(); // Black wins
167
168        // White is on move; Black has won → -WIN_SCORE from White's POV.
169        assert_eq!(evaluate(&game), -WIN_SCORE);
170    }
171}
