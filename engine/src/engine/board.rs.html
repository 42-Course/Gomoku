1//! Bit-packed board storage and line iteration.
2//!
3//! Stones are stored in two per-player [`BitBoard`]s rather than a 2D array
4//! of [`Cell`]s. The board exposes two complementary views:
5//!
6//! - **Cell-level** ([`Board::cell_at`], [`Board::is_empty`], …) for rule
7//!   logic that thinks one square at a time.
8//! - **Line-level** ([`Board::pack_line`], [`Board::for_each_line`]) for
9//!   pattern detection that thinks one row/column/diagonal at a time.
10//!
11//! The line view packs up to 19 cells of one direction into a pair of
12//! `u32` bitmasks `(me, opp)`. Off-the-board positions are zero in *both*
13//! masks, which makes them walls - see the [`crate::patterns`] module for
14//! how that interacts with pattern recognition.
15use std::ops::{
16    BitAnd,
17    BitAndAssign,
18    BitOr,
19    BitOrAssign,
20    BitXor,
21    BitXorAssign,
22    Not,
23    Shl,
24    Shr,
25};
26use std::sync::LazyLock;
27use crate::constants::{BOARD_SIZE, BOARD_SIZE_I, CELL_COUNT};
28use crate::game::{Direction, Cell, Player, Pos};
29use crate::patterns::has_stable_five;
30const WORD_COUNT: usize = CELL_COUNT.div_ceil(64);

32pub static LEFT_EDGE: LazyLock<BitBoard> =
33    LazyLock::new(BitBoard::left_edge);
34
35pub static RIGHT_EDGE: LazyLock<BitBoard> =
36    LazyLock::new(BitBoard::right_edge);

38/// One player's stones, stored as a bitmap with one bit per cell.
39///
40/// Cells are flattened in row-major order: bit `y * BOARD_SIZE + x` is set
41/// when that cell holds a stone. Six `u64` words is enough for 19×19 = 361
42/// cells with room to spare.
43#[derive(Copy, Clone, Debug, PartialEq, Eq)]
44pub struct BitBoard {
45    words: [u64; WORD_COUNT],
46}
47
48impl Default for BitBoard {
49    fn default() -> Self {
50        Self::new()
51    }
52}
53
54impl BitBoard {
55    /// Create an empty bitboard.
56    pub const fn new() -> Self {
57        Self {
58            words: [0; 6]
59        }
60    }
61    /// Set the bit at `idx`.
62    #[inline]
63    fn set(&mut self, idx: usize) {
64        let w = idx / 64;
65        let b = idx % 64;
66        self.words[w] |= 1u64 << b;
67    }
68
69    /// Clear the bit at `idx`.
70    #[inline]
71    fn clear(&mut self, idx: usize) {
72        let w = idx / 64;
73        let b = idx % 64;
74        self.words[w] &= !(1u64 << b);
75    }
76
77    /// Returns whether the bit at `idx` is set.
78    #[inline]
79    fn get(&self, idx: usize) -> bool {
80        let w = idx / 64;
81        let b = idx % 64;
82        (self.words[w] >> b) & 1 == 1
83    }
84
85    /// Map `(x, y)` to a flat bit index in row-major order.
86    #[allow(dead_code)]
87    #[inline]
88    fn index(x: usize, y: usize) -> usize {
89        y * BOARD_SIZE + x
90    }
91
92    /// Mark cell `(x, y)` as containing a stone.
93    #[inline]
94    pub fn place_stone(&mut self, pos: Pos) {
95        self.set(pos.idx());
96    }
97
98    /// Clear the stone at cell `(x, y)`.
99    #[inline]
100    pub fn remove_stone(&mut self, pos: Pos) {
101        self.clear(pos.idx());
102    }
103
104    /// Whether cell `(x, y)` holds a stone.
105    #[inline]
106    pub fn is_occupied(&self, pos: Pos) -> bool {
107        self.get(pos.idx())
108    }
109
110    /// Create a mask containing all cells on the left edge (`x = 0`)
111    /// of the board.
112    ///
113    /// Used to prevent horizontal and diagonal wraparound when
114    /// shifting bitboards westward.
115    pub fn left_edge() -> Self {
116        let mut bb = Self::new();
117
118        for y in 0..BOARD_SIZE {
119            bb.set(y * BOARD_SIZE);
120        }
121
122        bb
123    }
124
125    /// Create a mask containing all cells on the right edge
126    /// (`x = BOARD_SIZE - 1`) of the board.
127    ///
128    /// Used to prevent horizontal and diagonal wraparound when
129    /// shifting bitboards eastward.
130    pub fn right_edge() -> Self {
131        let mut bb = Self::new();
132
133        for y in 0..BOARD_SIZE {
134            bb.set(y * BOARD_SIZE + BOARD_SIZE - 1);
135        }
136
137        bb
138    }
139
140    /// Shift all bits one cell to the east (+x).
141    ///
142    /// Bits on the right edge are discarded to prevent
143    /// wraparound into the next row.
144    #[inline]
145    pub fn east(self) -> Self {
146        (self & !*RIGHT_EDGE) << 1
147    }
148
149    /// Shift all bits one cell to the west (-x).
150    ///
151    /// Bits on the left edge are discarded to prevent
152    /// wraparound into the previous row.
153    #[inline]
154    pub fn west(self) -> Self {
155        (self & !*LEFT_EDGE) >> 1
156    }
157
158    /// Shift all bits one cell to the south (+y).
159    #[inline]
160    pub fn south(self) -> Self {
161        self << BOARD_SIZE
162    }
163
164    /// Shift all bits one cell to the north (-y).
165    #[inline]
166    pub fn north(self) -> Self {
167        self >> BOARD_SIZE
168    }
169
170    /// Shift all bits one cell to the south-east (+x, +y).
171    ///
172    /// Bits on the right edge are discarded before shifting
173    /// to prevent diagonal wraparound.
174    #[inline]
175    pub fn south_east(self) -> Self {
176        (self & !*RIGHT_EDGE) << (BOARD_SIZE + 1)
177    }
178
179    /// Shift all bits one cell to the south-west (-x, +y).
180    ///
181    /// Bits on the left edge are discarded before shifting
182    /// to prevent diagonal wraparound.
183    #[inline]
184    pub fn south_west(self) -> Self {
185        (self & !*LEFT_EDGE) << (BOARD_SIZE - 1)
186    }
187
188    /// Shift all bits one cell to the north-east (+x, -y).
189    ///
190    /// Bits on the right edge are discarded before shifting
191    /// to prevent diagonal wraparound.
192    #[inline]
193    pub fn north_east(self) -> Self {
194        (self & !*RIGHT_EDGE) >> (BOARD_SIZE - 1)
195    }
196
197    /// Shift all bits one cell to the north-west (-x, -y).
198    ///
199    /// Bits on the left edge are discarded before shifting
200    /// to prevent diagonal wraparound.
201    #[inline]
202    pub fn north_west(self) -> Self {
203        (self & !*LEFT_EDGE) >> (BOARD_SIZE + 1)
204    }
205
206    /// Shift the bitboard by one cell in the given direction.
207    ///
208    /// Edge masks are applied automatically to prevent
209    /// horizontal and diagonal wraparound.
210    #[allow(dead_code)]
211    #[inline]
212    pub fn shift(self, dir: Direction) -> Self {
213        match dir {
214            Direction::Horizontal => self.east(),
215
216            Direction::Vertical => self.south(),
217
218            Direction::Diagonal => self.south_east(),
219
220            Direction::AntiDiagonal => self.south_west(),
221        }
222    }
223
224    /// Returns whether at least one bit is set.
225    #[inline]
226    pub fn any(self) -> bool {
227        self.words.iter().any(|&w| w != 0)
228    }
229}

231/// Bitwise OR between two bitboards.
232impl BitOr for BitBoard {
233    type Output = Self;
234
235    fn bitor(self, rhs: Self) -> Self::Output {
236        let mut out = Self::new();
237
238        for i in 0..WORD_COUNT {
239            out.words[i] = self.words[i] | rhs.words[i];
240        }
241
242        out
243    }
244}
245
246/// In-place bitwise OR assignment.
247impl BitOrAssign for BitBoard {
248    fn bitor_assign(&mut self, rhs: Self) {
249        for i in 0..WORD_COUNT {
250            self.words[i] |= rhs.words[i];
251        }
252    }
253}

255/// Bitwise AND between two bitboards.
256impl BitAnd for BitBoard {
257    type Output = Self;
258
259    fn bitand(self, rhs: Self) -> Self::Output {
260        let mut out = Self::new();
261
262        for i in 0..WORD_COUNT {
263            out.words[i] = self.words[i] & rhs.words[i];
264        }
265
266        out
267    }
268}

273        for i in 0..WORD_COUNT {

295/// In-place bitwise XOR assignment.
296impl BitXorAssign for BitBoard {
297    fn bitxor_assign(&mut self, rhs: Self) {
298        for i in 0..WORD_COUNT {
299            self.words[i] ^=
300                rhs.words[i];
301        }
302    }

305/// Bitwise NOT with unused tail bits masked out.
306impl Not for BitBoard {
307    type Output = Self;
308
309    fn not(self) -> Self::Output {
310        let mut out = Self::new();
311
312        for i in 0..WORD_COUNT {
313            out.words[i] = !self.words[i];
314        }
315
316        // Mask unused tail bits
317        let last_bits =
318            CELL_COUNT - (64 * (WORD_COUNT - 1));
319
320        let mask = (1u64 << last_bits) - 1;
321
322        out.words[WORD_COUNT - 1] &= mask;
323
324        out
325    }
326}

328/// Raw left shift across the packed bitboard without edge masking.
329impl Shl<usize> for BitBoard {
330    type Output = Self;
331
332    fn shl(self, shift: usize) -> Self::Output {
333        if shift == 0 {
334            return self;
335        }
336
337        let w = shift / 64;
338        let b = shift % 64;
339
340        let mut out = Self::new();
341
342        for src in 0..WORD_COUNT {
343            let dst = src + w;
344
345            if dst >= WORD_COUNT {
346                break;
347            }
348
349            out.words[dst] |= self.words[src] << b;
350
351            if b != 0 && dst + 1 < WORD_COUNT {
352                out.words[dst + 1] |=
353                    self.words[src] >> (64 - b);
354            }
355        }
356        out
357    }
358}

360/// Raw right shift across the packed bitboard without edge masking.
361impl Shr<usize> for BitBoard {
362    type Output = Self;
363
364    fn shr(self, shift: usize) -> Self::Output {
365        if shift == 0 {
366            return self;
367        }
368
369        let w = shift / 64;
370        let b = shift % 64;
371
372        let mut out = Self::new();
373
374        for dst in 0..WORD_COUNT {
375            let src = dst + w;
376
377            if src >= WORD_COUNT {
378                break;
379            }
380
381            out.words[dst] |= self.words[src] >> b;
382
383            if b != 0 && src + 1 < WORD_COUNT {
384                out.words[dst] |=
385                    self.words[src + 1] << (64 - b);
386            }
387        }
388
389        out
390    }
391}

393/// The full game board: one [`BitBoard`] per player.
394///
395/// Indexed `boards[player.idx()]`; black is index 0 and white is index 1.
396#[derive(Clone)]
397pub struct Board {
398    boards: [BitBoard; 2],
399}
400
401impl Default for Board {
402    fn default() -> Self {
403        Self::new()
404    }
405}
406
407impl Board {
408    /// Create an empty board.
409    pub fn new() -> Self {
410        Self {
411            boards: [BitBoard::new(), BitBoard::new()]
412        }
413    }

415    /// Place a stone for `player` at `(x, y)`.
416    ///
417    /// No bounds or occupancy check is performed - callers should go
418    /// through [`Board::empty_check`] first.
419    #[inline]
420    pub fn place_stone(&mut self, pos: Pos, player: Player) {
421        self.boards[player.idx()].place_stone(pos);
422    }
423
424    /// Remove `player`'s stone at `(x, y)`. Used by undo and capture.
425    #[inline]
426    pub fn remove_stone(&mut self, pos: Pos, player: Player) {
427        self.boards[player.idx()].remove_stone(pos);
428    }
429
430    /// Check if the given coordinate contains the player
431    #[inline]
432    pub fn has(&self, pos: Pos, player: Player) -> bool {
433        self.boards[player.idx()].is_occupied(pos)
434    }
435
436    /// Whether `(x, y)` is empty for *both* players.
437    #[inline]
438    pub fn is_empty(&self, pos: Pos) -> bool {
439        !self.has(pos, Player::Black) && !self.has(pos, Player::White)
440    }
441
442    /// Returns the bitboard containing all stones belonging
443    /// to the given player.
444    #[allow(dead_code)]
445    #[inline]
446    pub fn bits(&self, player: Player) -> BitBoard {
447        self.boards[player.idx()]
448    }
449
450    /// Validate that `(x, y)` is a legal placement target.
451    ///
452    /// # Errors
453    ///
454    /// - `"Out of bounds"` if `x` or `y` is outside `0..BOARD_SIZE`.
455    /// - `"Cell already occupied"` if either player has a stone there.
456    pub fn empty_check(&self, pos: Pos) -> Result<(), &'static str> {
457        if pos.idx() >= CELL_COUNT {
458            return Err("Out of bounds");
459        }
460
461        if !self.is_empty(pos) {
462            return Err("Cell already occupied");
463        }
464
465        Ok(())
466    }
467
468    /// Look up the contents of cell `Pos`.
469    ///
470    /// Returns `Some(Player)` if a stone is present, `None` if empty.
471    pub fn cell_at(&self, pos: Pos) -> Cell {
472        if self.boards[Player::Black.idx()].is_occupied(pos) {
473            Some(Player::Black)
474        } else if self.boards[Player::White.idx()].is_occupied(pos) {
475            Some(Player::White)
476        } else {
477            None
478        }
479    }
480
481    /// Look up the contents of cell `(x, y)`.
482    ///
483    /// Returns `Some(Player)` if a stone is present, `None` if empty.
484    #[allow(dead_code)]
485    pub fn cell_at_xy(&self, x: usize, y: usize) -> Cell {
486        let pos = Pos::from_xy(x, y);
487        if self.boards[Player::Black.idx()].is_occupied(pos) {
488            Some(Player::Black)
489        } else if self.boards[Player::White.idx()].is_occupied(pos) {
490            Some(Player::White)
491        } else {
492            None
493        }
494    }

496    /// Whether every cell on the board is occupied (used to flag draws).
497    pub fn is_full(&self) -> bool {
498        let occupied = self.boards[Player::Black.idx()] | self.boards[Player::White.idx()];
499        let w = BOARD_SIZE / 64;
500        let b = BOARD_SIZE % 64;
501
502        for i in 0..w {
503            if occupied.words[i] != u64::MAX {
504                return false;
505            }
506        }
507
508        let last_mast = (1u64 << b) - 1;
509
510        occupied.words[w] == last_mast
511    }

513    /// Returns whether the player has a stable five-in-a-row.
514    pub fn check_five(&self, player: Player) -> bool {
515        let me = self.boards[player.idx()];
516        let opp = self.boards[player.opponent().idx()];
517        has_stable_five(me, opp)
518    }

520    /// Pack one straight line of cells into two bitmasks.
521    ///
522    /// Walks from `(x0, y0)` along `(dx, dy)` for at most `max_len` steps,
523    /// stopping at the board edge. Returns `(me, opp, len)` where:
524    ///
525    /// - bit `i` of `me`  is set when the i-th cell holds `player`,
526    /// - bit `i` of `opp` is set when it holds the opponent,
527    /// - `len` is the number of cells actually visited.
528    ///
529    /// `len <= 19 < 32`, so a `u32` is always wide enough.
530    ///
531    /// Cells beyond the line (off-board, or past `len`) sit at zero in
532    /// both masks - pattern matchers that need an *empty* cell at an
533    /// endpoint won't see them as empty, so the board edge is a wall.
534    ///
535    /// # Examples
536    ///
537    /// ```ignore
538    /// // Pack the row containing y = 9 from left to right, for Black.
539    /// let (me, opp, len) = board.pack_line(0, 9, 1, 0, BOARD_SIZE as u32, Player::Black);
540    /// ```
541    pub fn pack_line(
542        &self,
543        x0: isize,
544        y0: isize,
545        dx: isize,
546        dy: isize,
547        max_len: u32,
548        player: Player,
549    ) -> (u32, u32, u32) {
550        let me_board = &self.boards[player.idx()];
551        let opp_board =
552            &self.boards[player.opponent().idx()];
553
554        let mut me = 0u32;
555        let mut opp = 0u32;
556        let mut len = 0u32;
557
558        let mut x = x0;
559        let mut y = y0;
560
561        while len < max_len
562            && x >= 0
563            && y >= 0
564            && x < BOARD_SIZE_I
565            && y < BOARD_SIZE_I
566        {
567            let idx =
568                (y as usize) * BOARD_SIZE
569                + (x as usize);
570
571            if me_board.get(idx) {
572                me |= 1 << len;
573            } else if opp_board.get(idx) {
574                opp |= 1 << len;
575            }
576
577            x += dx;
578            y += dy;
579            len += 1;
580        }
581
582        (me, opp, len)
583    }
584
585    /// Visit every distinct horizontal/vertical/diagonal/anti-diagonal line
586    /// on the board for `player`, calling `f(me, opp, len)` for each.
587    ///
588    /// Each line is packed via [`Board::pack_line`]. Lines shorter than
589    /// `min_len` are skipped (no useful pattern fits in fewer cells, and
590    /// skipping cuts the diagonal count almost in half).
591    ///
592    /// # Coverage
593    ///
594    /// - All `BOARD_SIZE` rows and `BOARD_SIZE` columns.
595    /// - Both diagonal families, starting from cells where the previous
596    ///   cell is off-board.
597    ///
598    /// # Examples
599    ///
600    /// ```ignore
601    /// let mut totals = PatternCounts::default();
602    /// board.for_each_line(Player::Black, 5, |me, opp, len| {
603    ///     totals.add(&count_patterns(me, opp, len));
604    /// });
605    /// ```
606    pub fn for_each_line<F>(&self, player: Player, min_len: u32, mut f: F)
607    where
608        F: FnMut(u32, u32, u32),
609    {
610        let n = BOARD_SIZE_I;
611
612        // Rows, then columns: always full board-sized lines.
613        for y in 0..n {
614            let (me, opp, len) = self.pack_line(0, y, 1, 0, BOARD_SIZE as u32, player);
615            if len >= min_len {
616                f(me, opp, len);
617            }
618        }
619        for x in 0..n {
620            let (me, opp, len) = self.pack_line(x, 0, 0, 1, BOARD_SIZE as u32, player);
621            if len >= min_len {
622                f(me, opp, len);
623            }
624        }
625
626        // Diagonals (dx=1, dy=1): start where the previous cell is off-board,
627        // i.e. x == 0 or y == 0.
628        for x in 0..n {
629            let (me, opp, len) = self.pack_line(x, 0, 1, 1, BOARD_SIZE as u32, player);
630            if len >= min_len {
631                f(me, opp, len);
632            }
633        }
634        for y in 1..n {
635            let (me, opp, len) = self.pack_line(0, y, 1, 1, BOARD_SIZE as u32, player);
636            if len >= min_len {
637                f(me, opp, len);
638            }
639        }
640
641        // Anti-diagonals (dx=1, dy=-1): start where x == 0 or y == n-1.
642        for x in 0..n {
643            let (me, opp, len) = self.pack_line(x, n - 1, 1, -1, BOARD_SIZE as u32, player);
644            if len >= min_len {
645                f(me, opp, len);
646            }
647        }
648        for y in 0..(n - 1) {
649            let (me, opp, len) = self.pack_line(0, y, 1, -1, BOARD_SIZE as u32, player);
650            if len >= min_len {
651                f(me, opp, len);
652            }
653        }
654    }

656    /// Print the board to stdout in human-readable form.
657    ///
658    /// Black is shown as `X`, White as `O`, and empty cells as `.`. Used
659    /// for debugging and the [`crate::play!`] macro.
660    pub fn print_board(&self) {
661        let height = BOARD_SIZE;
662        let width = BOARD_SIZE;
663        let row_digits = height.to_string().len();
664
665        print!("{}", " ".repeat(row_digits + 3));
666
667        for col in 0..width {
668            let tens = col / 10;
669            if tens > 0 {
670                print!("{} ", tens);
671            } else {
672                print!("  ");
673            }
674        }
675
676        println!();
677
678        print!("{}", " ".repeat(row_digits + 3));
679
680        for col in 0..width {
681            print!("{} ", col % 10);
682        }
683
684        println!();
685        println!("{}{}", " ".repeat(row_digits + 3), "-".repeat(width * 2));
686
687        for y in 0..height {
688            let mut row_str = String::new();
689
690            for x in 0..width {
691                let pos = Pos::from_xy(x, y);
692                let symbol = match self.cell_at(pos) {
693                    None => ".",
694                    Some(Player::Black) => "X",
695                    Some(Player::White) => "O",
696                };
697
698                row_str.push_str(symbol);
699                row_str.push(' ');
700            }
701
702            println!("{:2} | {}", y, row_str.trim_end());
703        }
704    }
