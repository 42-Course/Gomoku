1//! Bitwise pattern detection on a single packed line.
2//!
3//! A "line" is one row, column, or diagonal of the board, packed as two
4//! `u32`s, `me` (the bits of the player we're scoring) and `opp` (their
5//! opponent). Bit `i` of each word corresponds to cell `i` along the line.
6//! `len` says how many of the low bits are part of the line; the rest are
7//! zero in both words and therefore behave as "off-board" (a wall: not me,
8//! not opp, not empty).
9//!
10//! That convention is the whole reason this module is short: a pattern
11//! requiring an *empty* cell at an endpoint won't match against a board
12//! edge, the bit there is 0 in `empty()`, and a pattern requiring our
13//! stone won't see one off-board either. Edges fall out for free.
14//!
15//! # Public API
16//!
17//! - [`count_patterns`] — tally every maximal run on a line by length and
18//!   openness.
19//! - [`has_free_three`] — yes/no check for the four free-three shapes,
20//!   used by the double-three rule.
21//! - [`PatternCounts`] — the tally type produced by [`count_patterns`].
22

25/// Tally of distinct max-length runs found on one line for one player.
26///
27/// "Max-length" means each entry counts a run by its longest extent: a
28/// 5-run is *not* also counted as several 4-runs.
29///
30/// "Open" runs have empty cells on both sides; "closed" runs have an
31/// opponent stone or the board edge on exactly one side.
32#[derive(Debug, Default, Clone, Copy, PartialEq, Eq)]
33pub struct PatternCounts {
34    /// Runs of 5 or more in a row.
35    pub fives: u32,
36    /// 4-runs with empty cells on both sides.
37    pub open_four: u32,
38    /// 4-runs with one side blocked.
39    pub closed_four: u32,
40    /// 3-runs with empty cells on both sides.
41    pub open_three: u32,
42    /// 3-runs with one side blocked.
43    pub closed_three: u32,
44    /// 2-runs with empty cells on both sides.
45    pub open_two: u32,
46    /// 2-runs with one side blocked.
47    pub closed_two: u32,
48}
49
50impl PatternCounts {
51    /// Add the counts in `rhs` to `self` field-by-field.
52    pub fn add(&mut self, rhs: &PatternCounts) {
53        self.fives += rhs.fives;
54        self.open_four += rhs.open_four;
55        self.closed_four += rhs.closed_four;
56        self.open_three += rhs.open_three;
57        self.closed_three += rhs.closed_three;
58        self.open_two += rhs.open_two;
59        self.closed_two += rhs.closed_two;
60    }
61}

63#[inline]
64fn line_mask(len: u32) -> u32 {
65    if len >= 32 { u32::MAX } else { (1u32 << len) - 1 }
66}
67
68/// Bits set at position `i` iff a maximal run of `me` of *exactly* length
69/// `k` starts at `i`. "Maximal" = the cell before (`i-1`) and after
70/// (`i+k`) are not `me`, where off-the-line counts as not-me too.
71#[inline]
72fn run_starts(m: u32, k: u32, mask: u32) -> u32 {
73    let mut consecutive = m & mask;
74    for s in 1..k {
75        consecutive &= m >> s;
76    }
77    consecutive &= mask;
78
79    let not_left = !(m << 1);
80    let not_right = !(m >> k);
81    consecutive & not_left & not_right & mask
82}

84/// Walk one packed line and tally every distinct run by length and openness.
85///
86/// Each maximal run contributes to exactly one bucket — a 5-run is a
87/// `fives`, *not* additionally a `closed_four` plus a `closed_three`. Runs
88/// shorter than two cells are ignored.
89///
90/// # Arguments
91///
92/// - `me`  — bits of the player whose patterns we're tallying.
93/// - `opp` — bits of the opponent.
94/// - `len` — number of cells in the packed line (`<= 19`).
95///
96/// # Examples
97///
98/// ```ignore
99/// // ".XXXX." has one open four.
100/// let me  = 0b011110;
101/// let opp = 0;
102/// let counts = count_patterns(me, opp, 6);
103/// assert_eq!(counts.open_four, 1);
104/// ```
105pub fn count_patterns(me: u32, opp: u32, len: u32) -> PatternCounts {
106    let mask = line_mask(len);
107    let m = me & mask;
108    let o = opp & mask;
109    let e = !(m | o) & mask;
110
111    // Anything 5+ in a row counts as a five, a single bit is enough.
112    let five_or_more = (m & (m >> 1) & (m >> 2) & (m >> 3) & (m >> 4)) & mask;
113    let fives = five_or_more.count_ones();
114
115    let four = run_starts(m, 4, mask);
116    let three = run_starts(m, 3, mask);
117    let two = run_starts(m, 2, mask);
118
119    let split = |run: u32, k: u32| -> (u32, u32) {
120        // bit i of (e << 1) holds e[i-1]; bit i of (e >> k) holds e[i+k].
121        let left_open = (e << 1) & mask;
122        let right_open = (e >> k) & mask;
123        let both_open = run & left_open & right_open;
124        let one_open = run & (left_open ^ right_open);
125        (both_open.count_ones(), one_open.count_ones())
126    };
127
128    let (open_four, closed_four) = split(four, 4);
129    let (open_three, closed_three) = split(three, 3);
130    let (open_two, closed_two) = split(two, 2);
131
132    PatternCounts {
133        fives,
134        open_four,
135        closed_four,
136        open_three,
137        closed_three,
138        open_two,
139        closed_two,
140    }
141}

143/// Does this packed line contain *any* free-three pattern for `me`?
144///
145/// Free threes are the patterns that turn into an open four with one move:
146/// `.XXX..`, `..XXX.`, `.XX.X.`, `.X.XX.` (each in a 6-cell window with
147/// empty endpoints). The Gomoku double-three rule needs to know whether
148/// playing a stone has *created* one of these along a given direction.
149pub fn has_free_three(me: u32, opp: u32, len: u32) -> bool {
150    let mask = line_mask(len);
151    let m = me & mask;
152    let o = opp & mask;
153    let e = !(m | o) & mask;
154
155    // Each pattern is a 6-cell window. Bit i of the result means the
156    // pattern starts at position i, so we need every shift to align.
157    let solid_l = e & (m >> 1) & (m >> 2) & (m >> 3) & (e >> 4) & (e >> 5);
158    let solid_r = e & (e >> 1) & (m >> 2) & (m >> 3) & (m >> 4) & (e >> 5);
159    let split_l = e & (m >> 1) & (m >> 2) & (e >> 3) & (m >> 4) & (e >> 5);
160    let split_r = e & (m >> 1) & (e >> 2) & (m >> 3) & (m >> 4) & (e >> 5);
161
162    ((solid_l | solid_r | split_l | split_r) & mask) != 0
163}

169    /// Build a line from a string of `X`/`O`/`.` characters, returning
170    /// (me, opp, len). `X` is me.
171    fn line(s: &str) -> (u32, u32, u32) {
172        let mut me = 0u32;
173        let mut opp = 0u32;
174        for (i, c) in s.chars().enumerate() {
175            match c {
176                'X' => me |= 1 << i,
177                'O' => opp |= 1 << i,
178                '.' => {}
179                other => panic!("bad cell '{other}' in line {s:?}"),
180            }
181        }
182        (me, opp, s.chars().count() as u32)
183    }

185    #[test]
186    fn five_in_a_row_is_a_five() {
187        let (m, o, l) = line(".XXXXX.");
188        let c = count_patterns(m, o, l);
189        assert_eq!(c.fives, 1);
190        // The 5-run isn't double-counted as smaller runs.
191        assert_eq!(c.open_four, 0);
192        assert_eq!(c.closed_four, 0);
193    }
