1//! Bitwise pattern detection on a single packed line.
2//!
3//! A "line" is one row, column, or diagonal of the board, packed as two
4//! `u32`s, `me` (the bits of the player we're scoring) and `opp` (their
5//! opponent). Bit `i` of each word corresponds to cell `i` along the line.
6//! `len` says how many of the low bits are part of the line; the rest are
7//! zero in both words and therefore behave as "off-board" (a wall: not me,
8//! not opp, not empty).
9//!
10//! That convention is the whole reason this module is short: a pattern
11//! requiring an *empty* cell at an endpoint won't match against a board
12//! edge, the bit there is 0 in `empty()`, and a pattern requiring our
13//! stone won't see one off-board either. Edges fall out for free.
14//!
15//! # Public API
16//!
17//! Packed-line (`u32`) helpers:
18//!
19//! - [`count_patterns`] — tally every maximal run on a line by length and
20//!   openness.
21//! - [`has_free_three`] — yes/no check for the four free-three shapes,
22//!   used by the double-three rule.
23//! - [`PatternCounts`] — the tally type produced by [`count_patterns`].
24//!
25//! Whole-board ([`BitBoard`]) helpers:
26//!
27//! - [`five_mask`] — starting cells of every contiguous five along a
28//!   direction.
29//! - [`expand_five`] — expand five-start bits into full five-cell masks.
30//! - [`capturable_mask`] — every player stone that is part of a
31//!   capturable pair in any direction.
32//! - [`has_stable_five`] — does the player have a five whose stones
33//!   can't all be broken by a capture?
34

40/// Tally of distinct max-length runs found on one line for one player.
41///
42/// "Max-length" means each entry counts a run by its longest extent: a
43/// 5-run is *not* also counted as several 4-runs.
44///
45/// "Open" runs have empty cells on both sides; "closed" runs have an
46/// opponent stone or the board edge on exactly one side.
47#[derive(Debug, Default, Clone, Copy, PartialEq, Eq)]
48pub struct PatternCounts {
49    /// Runs of 5 or more in a row.
50    pub fives: u32,
51    /// 4-runs with empty cells on both sides.
52    pub open_four: u32,
53    /// 4-runs with one side blocked.
54    pub closed_four: u32,
55    /// 3-runs with empty cells on both sides.
56    pub open_three: u32,
57    /// 3-runs with one side blocked.
58    pub closed_three: u32,
59    /// 2-runs with empty cells on both sides.
60    pub open_two: u32,
61    /// 2-runs with one side blocked.
62    pub closed_two: u32,
63}
64
65impl PatternCounts {
66    /// Add the counts in `rhs` to `self` field-by-field.
67    pub fn add(&mut self, rhs: &PatternCounts) {
68        self.fives += rhs.fives;
69        self.open_four += rhs.open_four;
70        self.closed_four += rhs.closed_four;
71        self.open_three += rhs.open_three;
72        self.closed_three += rhs.closed_three;
73        self.open_two += rhs.open_two;
74        self.closed_two += rhs.closed_two;
75    }
76}

78#[inline]
79fn line_mask(len: u32) -> u32 {
80    if len >= 32 { u32::MAX } else { (1u32 << len) - 1 }
81}
82
83/// Bits set at position `i` iff a maximal run of `me` of *exactly* length
84/// `k` starts at `i`. "Maximal" = the cell before (`i-1`) and after
85/// (`i+k`) are not `me`, where off-the-line counts as not-me too.
86#[inline]
87fn run_starts(m: u32, k: u32, mask: u32) -> u32 {
88    let mut consecutive = m & mask;
89    for s in 1..k {
90        consecutive &= m >> s;
91    }
92    consecutive &= mask;
93
94    let not_left = !(m << 1);
95    let not_right = !(m >> k);
96    consecutive & not_left & not_right & mask
97}

99/// Walk one packed line and tally every distinct run by length and openness.
100///
101/// Each maximal run contributes to exactly one bucket — a 5-run is a
102/// `fives`, *not* additionally a `closed_four` plus a `closed_three`. Runs
103/// shorter than two cells are ignored.
104///
105/// # Arguments
106///
107/// - `me`  — bits of the player whose patterns we're tallying.
108/// - `opp` — bits of the opponent.
109/// - `len` — number of cells in the packed line (`<= 19`).
110///
111/// # Examples
112///
113/// ```ignore
114/// // ".XXXX." has one open four.
115/// let me  = 0b011110;
116/// let opp = 0;
117/// let counts = count_patterns(me, opp, 6);
118/// assert_eq!(counts.open_four, 1);
119/// ```
120pub fn count_patterns(me: u32, opp: u32, len: u32) -> PatternCounts {
121    let mask = line_mask(len);
122    let m = me & mask;
123    let o = opp & mask;
124    let e = !(m | o) & mask;
125
126    // Anything 5+ in a row counts as a five, a single bit is enough.
127    let five_or_more = (m & (m >> 1) & (m >> 2) & (m >> 3) & (m >> 4)) & mask;
128    let fives = five_or_more.count_ones();
129
130    let four = run_starts(m, 4, mask);
131    let three = run_starts(m, 3, mask);
132    let two = run_starts(m, 2, mask);
133
134    let split = |run: u32, k: u32| -> (u32, u32) {
135        // bit i of (e << 1) holds e[i-1]; bit i of (e >> k) holds e[i+k].
136        let left_open = (e << 1) & mask;
137        let right_open = (e >> k) & mask;
138        let both_open = run & left_open & right_open;
139        let one_open = run & (left_open ^ right_open);
140        (both_open.count_ones(), one_open.count_ones())
141    };
142
143    let (open_four, closed_four) = split(four, 4);
144    let (open_three, closed_three) = split(three, 3);
145    let (open_two, closed_two) = split(two, 2);
146
147    PatternCounts {
148        fives,
149        open_four,
150        closed_four,
151        open_three,
152        closed_three,
153        open_two,
154        closed_two,
155    }
156}

158/// Does this packed line contain *any* free-three pattern for `me`?
159///
160/// Free threes are the patterns that turn into an open four with one move:
161/// `.XXX..`, `..XXX.`, `.XX.X.`, `.X.XX.` (each in a 6-cell window with
162/// empty endpoints). The Gomoku double-three rule needs to know whether
163/// playing a stone has *created* one of these along a given direction.
164pub fn has_free_three(me: u32, opp: u32, len: u32) -> bool {
165    let mask = line_mask(len);
166    let m = me & mask;
167    let o = opp & mask;
168    let e = !(m | o) & mask;
169
170    // Each pattern is a 6-cell window. Bit i of the result means the
171    // pattern starts at position i, so we need every shift to align.
172    let solid_l = e & (m >> 1) & (m >> 2) & (m >> 3) & (e >> 4) & (e >> 5);
173    let solid_r = e & (e >> 1) & (m >> 2) & (m >> 3) & (m >> 4) & (e >> 5);
174    let split_l = e & (m >> 1) & (m >> 2) & (e >> 3) & (m >> 4) & (e >> 5);
175    let split_r = e & (m >> 1) & (e >> 2) & (m >> 3) & (m >> 4) & (e >> 5);
176
177    ((solid_l | solid_r | split_l | split_r) & mask) != 0
178}

180/// Returns the starting cells of every contiguous
181/// five alignment in the given direction.
182#[inline]
183pub fn five_mask(
184    bits: BitBoard,
185    dir: Direction,
186) -> BitBoard {
187    let run2 = bits & dir.backward(bits);
188    let run3 = run2 & dir.backward(run2);
189    let run4 = run3 & dir.backward(run3);
190    run4 & dir.backward(run4)
191}

193/// Expands five starting positions into full five-cell masks.
194#[inline]
195pub fn expand_five(
196    starts: BitBoard,
197    dir: Direction,
198) -> BitBoard {
199    let b = dir.forward(starts);
200    let c = dir.forward(b);
201    let d = dir.forward(c);
202    let e = dir.forward(d);
203
204    starts | b | c | d | e
205}

207/// Returns whether the player has a stable
208/// five-in-a-row alignment.
209///
210/// A five is "stable" if no stone within its five cells can be
211/// captured by the opponent. When a direction contains multiple
212/// overlapping fives (e.g. a six-in-a-row contains two), each
213/// candidate is checked individually — a five whose own five
214/// cells don't intersect any capturable stone wins even if a
215/// sibling five in the same direction is unstable.
216pub fn has_stable_five(
217    me: BitBoard,
218    opp: BitBoard,
219) -> bool {
220    let capturable = capturable_mask(me, opp);
221
222    for dir in Direction::all() {
223        let starts = five_mask(me, dir);
224
225        if !starts.any() {
226            continue;
227        }
228
229        // Dilate `capturable` backward 0..=4 steps so that every
230        // start position whose five-cell window covers a capturable
231        // stone gets a bit set. The complement intersected with
232        // `starts` is the set of stable five starts.
233        let mut killed = capturable;
234        let mut acc = capturable;
235        for _ in 0..4 {
236            acc = dir.backward(acc);
237            killed |= acc;
238        }
239
240        if (starts & !killed).any() {
241            return true;
242        }
243    }
244
245    false
246}

248/// Returns all stones that are currently capturable by the opponent.
249pub fn capturable_mask(
250    me: BitBoard,
251    opp: BitBoard,
252) -> BitBoard {
253    let empty = !(me | opp);
254
255    let mut out = BitBoard::new();
256    for dir in Direction::all() {
257        out |= capturable_pairs_dir(
258            me,
259            opp,
260            empty,
261            dir,
262        );
263    }
264    out
265}
266
267/// Returns all capturable pairs in one direction.
268///
269/// Each bit `i` in the result marks a player stone that is part
270/// of a capturable pair along `dir`. The two arms below match the
271/// pair against position `i` (the *forward* stone of the pair),
272/// then `dir.backward(starts)` adds the *backward* stone of each
273/// matched pair.
274#[inline]
275fn capturable_pairs_dir(
276    player: BitBoard,
277    opponent: BitBoard,
278    empty: BitBoard,
279    dir: Direction
280) -> BitBoard {
281    //
282    // . X X O   (empty backward of pair, opponent forward of pair)
283    //
284    let left =
285        player &
286        dir.forward(player) &
287        dir.forward(dir.forward(empty)) &
288        dir.backward(opponent);
289
290    //
291    // O X X .   (opponent backward of pair, empty forward of pair)
292    //
293    let right =
294        player &
295        dir.forward(player) &
296        dir.backward(empty) &
297        dir.forward(dir.forward(opponent));
298
299    let starts = left | right;
300
301    starts | dir.backward(starts)
302}

310    /// Build a line from a string of `X`/`O`/`.` characters, returning
311    /// (me, opp, len). `X` is me.
312    fn line(s: &str) -> (u32, u32, u32) {
313        let mut me = 0u32;
314        let mut opp = 0u32;
315        for (i, c) in s.chars().enumerate() {
316            match c {
317                'X' => me |= 1 << i,
318                'O' => opp |= 1 << i,
319                '.' => {}
320                other => panic!("bad cell '{other}' in line {s:?}"),
321            }
322        }
323        (me, opp, s.chars().count() as u32)
324    }

393    fn bb(coords: &[(usize, usize)]) -> BitBoard {
394        let mut out = BitBoard::new();
395
396        for &(x, y) in coords {
397            out.place_stone(Pos::from_xy(x, y));
398        }
399
400        out
401    }
